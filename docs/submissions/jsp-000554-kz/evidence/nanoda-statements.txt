def JSP000554.primorial (n : Nat) : Nat := Finset.prod (Finset.filter Nat.Prime (Finset.range n)) (fun (a : Nat) => a)

def JSP000554.omegaSet (n : Nat) : Finset Nat :=
  Finset.filter

        (forall (j : Nat),
